R0L     EQU  $0
R0H     EQU  $1
R14L    EQU  $1C
R14H    EQU  $1D
R15L    EQU  $1E
R15H    EQU  $1F
SAVE    EQU  $FF4A
RESTORE EQU  $FF3F

        ORG  $F689

        AST  32

        JSR  SAVE           ;PRESERVE 6502 REG CONTENTS
        PLA
        STA  R15L           ;INIT SWEET16 PC
        PLA                 ;FROM RETURN
        STA  R15H           ;ADDRESS
SW16B   JSR  SW16C          ;INTERPRET AND EXECUTE
        JMP  SW16B          ;ONE SWEET16 INSTR.

SW16C   LDA  >SET           ;COMMON HIGH BYTE FOR ALL ROUTINES
        PHA                 ;PUSH ON STACK FOR RTS
        INC  R15L           ; 5  
        BNE  GETB2          ; 3  INCR SWEET16 PC FOR FETCH
        INC  R15H           ;   
GETB2   LDY  $0             ; 2  
        LDA  (R15L),Y       ; 5  FETCH BYTE
        AND  $F             ;MASK REG SPECIFICATION
        ASL                 ;DOUBLE FOR TWO BYTE REGISTERS
        TAX                 ;TO X REG FOR INDEXING
        LSR
        EOR  (R15L),Y       ;NOW HAVE OPCODE
        BEQ  TOBR           ;IF ZERO THEN NON-REG OP
        STY  R14H           ;INDICATE "PRIOR RESULT REG" ZERO
        LSR
        LSR                 ;OPCODE*2 TO LSB'S
        LSR
        TAY                 ;TO Y REG FOR INDEXING
        LDA  OPTBL-2,Y      ;LOW ORDER ADR BYTE
        PHA                 ;ONTO STACK
        RTS                 ;GOTO REG-OP ROUTINE -->  CARRY  CLEAR

TOBR    INC  R15L
        BNE  TOBR2          ;INCR PC --> SKIP OPCODE
        INC  R15H
TOBR2   LDA  BRTBL,X        ;LOW ORDER ADR BYTE
        PHA                 ;ONTO STACK FOR NON-REG OP
        LDA  R14H           ;"PRIOR RESULT REG" INDEX
        LSR                 ;PREPARE CARRY FOR BC, BNC.
        RTS                 ;GOTO NON-REG OP ROUTINE

RTNZ    PLA                 ;POP RETURN ADDRESS
        PLA
        JSR  RESTORE        ;RESTORE 6502 REG CONTENTS
        JMP  (R15L)         ;RETURN TO 6502 CODE VIA PC

SETZ    DEY                 ; Y BEFORE = 2
        LDA  (R15L),Y       ;FETCH HI BYTE
        PHA
        DEY                 ; Y = 0
        LDA  (R15L),Y       ;FETCH LO BYTE
        TAY
        CPX  #30            ; IS REG PC
        BEQ   STZPOP        ; THEN SKIP INCREMENTING PC
        STX  R14H           ;INDICATE RESULT
        LDA  R15L
        ADC   #2
        STA  R15L
        BCC   STZPOP
        INC  R15H           
STZPOP  PLA
STZ     STY  R0L,X          ;MOVE TO RX
        STA  R0H,X
        RTS

LDDAT2  JSR  LDAT2
        TAY
LDAT2   LDA  (R0L,X)        ;LOAD INDIRECT (RX)
INRZ    INC  R0L,X          ;  ZERO
        BNE  INR2           ;INCR RX
        INC  R0H,X
INR2    RTS
                         
BRTBL   DFB  RTN-1          ;0
OPTBL   DFB  SET-1          ;1X
        DFB  BR-1           ;1
        DFB  LD-1           ;2X
        DFB  BNC-1          ;2
        DFB  ST-1           ;3X
        DFB  BC-1           ;3
        DFB  LDAT-1         ;4X
        DFB  BP-1           ;4
        DFB  STAT-1         ;5X
        DFB  BM-1           ;5
        DFB  LDDAT-1        ;6X
        DFB  BZ-1           ;6
        DFB  STDAT-1        ;7X
        DFB  BNZ-1          ;7
        DFB  POP-1          ;8X
        DFB  BM1-1          ;8
        DFB  STPAT-1        ;9X
        DFB  BNM1-1         ;9
        DFB  ADD-1          ;AX
        DFB  BK-1           ;A
        DFB  SUB-1          ;BX
        DFB  RS-1           ;B
        DFB  POPD-1         ;CX
        DFB  BS-1           ;C
        DFB  CPR-1          ;DX
        DFB  NUL-1          ;D
        DFB  INR-1          ;EX
        DFB  NUL-1          ;E
        DFB  DCR-1          ;FX
        DFB  NUL-1          ;F

* FOLLOWING CODE MUST BE
* CONTAINED ON A SINGLE PAGE!

SET     BPL  SETZ           ;ALWAYS TAKEN
LD      LDY  R0L,X
BK      EQU  *-1
        LDA  R0H,X          ;MOVE RX TO R0
STAY    STY  R0L
        STA  R0H
        RTS

ST      LDY  R0L
        LDA  R0H
        BCC  STZ

INR     STX  R14H
        BCC   INRZ

STAT    LDA  R0L
STAT2   STA  (R0L,X)        ;STORE BYTE INDIRECT
        JMP  INRZ  
         
LDAT    LDY  $0             ;ZERO HIGH ORDER R0 BYTE
LDATZ   JSR  LDAT2          ;LOAD INDIRECT (RX)
        JMP  STYA           ;TO R0

POP     LDY  $0             ;HIGH ORDER BYTE = 0
POP2    JSR  DCRZ           ;DECR RX  ;LOW ORDER BYTE
        LDA  (R0L,X)
STYA    STA  R0L            ;TO R0
        STY  R0H
        RTS

POPD    JSR  DCRZ           ;DECR RX ;POP HIGH ORDER BYTE @RX
        LDA  (R0L,X)
        TAY                 ;SAVE IN Y REG
        BCC  POP2

LDDAT   JSR  LDDAT2         ; WORD TO R0, INCR RX
        JMP  STAY

STDAT   JSR  STAT           ;STORE INDIRECT LOW ORDER
        LDA  R0H            ;BYTE AND INCR RX. THEN
        BCC  STAT2          ;STORE HIGH ORDER BYTE. ;INCR RX AND RETURN

STPAT   JSR  DCRZ           ;DECR RX
        LDA  R0L
        STA  (R0L,X)        ;STORE R0 LOW BYTE @RX
        RTS

DCR     STX  R14H
DCRZ    LDA  R0L,X
        BNE  DCR2           ;DECR RX  AND FETHCH BYTE @RX
        DEC  R0H,X
DCR2    DEC  R0L,X
        RTS

SUB     LDY  $0             ;RESULT TO R0
CPR     SEC                 ;NOTE Y REG = 13*2 FOR CPR
        LDA  R0L
        SBC  R0L,X
        STA  R0L,Y          ;R0-RX TO RY
        LDA  R0H
        SBC  R0H,X
SUB2    STA  R0H,Y
        TYA                 ;LAST RESULT REG*2
        ADC  $0             ;CARRY TO LSB
        STA  R14H
        RTS

ADD     LDA  R0L
        ADC  R0L,X
        STA  R0L            ;R0+RX TO R0
        LDA  R0H
        ADC  R0H,X
        LDY  $0             ;R0 FOR RESULT
        BEQ  SUB2           ;FINISH ADD

BS      JSR  DCRZ           ;PUSH LOW PC BYTE VIA R12
        LDA  R15H
        STA  (R0L,X)
        JSR  DCRZ           ;PUSH LOW PC BYTE VIA R12
        LDA  R15L           ;NOTE X REG IS 12*2!
        STA  (R0L,X)
BR      CLC
BNC     BCS  BNC2           ;NO CARRY TEST
BR1     LDA  (R15L),Y       ;DISPLACEMENT BYTE
        BPL  BR2
        DEY
BR2     ADC  R15L           ;ADD TO PC
        STA  R15L
        TYA
        ADC  R15H
        STA  R15H
BNC2    RTS

BC      BCS  BR
        RTS

BP      ASL                 ;DOUBLE RESULT-REG INDEX
        TAX                 ;TO X REG FOR INDEXING
        LDA  R0H,X          ;TEST FOR PLUS
        BPL  BR1            ;BRANCH IF SO
        RTS

BM      ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0H,X          ;TEST FOR MINUS
        BMI  BR1
        RTS

BZ      ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X          ;TEST FOR ZERO
        ORA  R0H,X          ;(BOTH BYTES)
        BEQ  BR1            ;BRANCH IF SO
        RTS

BNZ     ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X          ;TEST FOR NON-ZERO
        ORA  R0H,X          ;(BOTH BYTES)
        BNE  BR1            ;BRANCH IF SO
        RTS

BM1     ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X          ;CHECK BOTH BYTES
        AND  R0H,X          ;FOR $FF (MINUS 1)
        EOR  $FF
        BEQ  BR1            ;BRANCH IF SO
        RTS

BNM1    ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X
        AND  R0H,X          ;CHECK BOTH BYTES FOR NO $FF
        EOR  $FF
        BNE  BR1            ;BRANCH IF NOT MINUS 1
NUL     RTS

RS      LDX  $18            ;12*2 FOR R12 AS STACK POINTER
        JSR  LDDAT2         ;DECR STACK POINTER
        STA  R15H
        STY  R15L
        RTS

RTN     JMP  RTNZ
