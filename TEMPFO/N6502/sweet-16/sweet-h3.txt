R0L     EQU  $0
R0H     EQU  $1
R14L    EQU  $1C
R14H    EQU  $1D
R15L    EQU  $1E
R15H    EQU  $1F
VECT    equ  $22
SAVE    EQU  $FF4A
RESTORE EQU  $FF3F

        ORG  $F689

        AST  32

        JSR  SAVE           ;PRESERVE 6502 REG CONTENTS
        LDA  >BRTBL         ;COMMON HIGH BYTE FOR OPTBL
        STA   VECT+1
        LDA  #4C            ;
        STA   VECT-1        ; JMP() CODE
        PLA
        STA  R15L           ;INIT SWEET16 PC
        PLA                 ;FROM RETURN
        STA  R15H           ;ADDRESS
        JSR  INCR15         ; INCR SWEET16 PC FOR FETCH
SW16A   JSR  SW16C          ; INTERPRET AND EXECUTE
        JMP  SW16A          ; ONE SWEET16 INSTR.

GETB    LDY  $0             ; 2  
GETBZ   LDA  (R15L),Y       ; 5  FETCH BYTE
INCR15  INC  R15L           ; 5  
        BNE  GETBZ2         ; 3  INCR SWEET16 PC FOR FETCH
        INC  R15H           ;   
GETBZ2  RTS

SW16C   JSR  GETB           ; INCR SWEET16 PC FOR FETCH
        STA  VECT           ;INDICATE "PRIOR RESULT REG" ZERO
        AND  $F             ;MASK REG SPECIFICATION
        ASL                 ;DOUBLE FOR TWO BYTE REGISTERS
        TAX                 ;TO X REG FOR INDEXING
        LSR
        EOR  VECT           ;NOW HAVE OPCODE
        BEQ  TOBR           ;IF ZERO THEN NON-REG OP

        STY  R14H           ;INDICATE "PRIOR RESULT REG" ZERO
        LSR
        LSR                 ;OPCODE*2 TO LSB'S
        LSR
        TAY                 ;TO Y REG FOR INDEXING
        ADC   #32
        STA   VECT
        JMP   VECT-1

TOBR    STX  VECT           ;"PRIOR RESULT REG" INDEX
        LDA  R14H           ;"PRIOR RESULT REG" INDEX
        LSR                 ;PREPARE CARRY FOR BC, BNC.
        JMP   VECT-1        ;GOTO NON-REG OP ROUTINE

RTN     PLA                 ;POP RETURN ADDRESS
        PLA
        JSR  RESTORE        ;RESTORE 6502 REG CONTENTS
        JMP  (R15L)         ;RETURN TO 6502 CODE VIA PC

SET     STX  R14H           ;INDICATE RESULT
        LDX   #PC
        JSR   LDDAT2        ;LOAD INDIRECT (PC)
        LDX  R14H           ;INDICATE RESULT
STZ     STY  R0L,X          ;MOVE TO RX
        STA  R0H,X
        RTS

LDDAT2  JSR  LDAT2          
        TAY
LDAT2   LDA  (R0L,X)        ;LOAD INDIRECT (RX)
INRZ    INC  R0L,X          ;  ZERO
        BNE  INR2           ;INCR RX
        INC  R0H,X
INR2    RTS


; THIS TABLE MUST BE ON START OF THE PAGE  (xx00h)
                     
BRTBL: 
      DW    RTN          ;0  
      DW    BR           ;1
      DW    BNC          ;2
      DW    BC           ;3
      DW    BP           ;4
      DW    BM           ;5
      DW    BZ           ;6
      DW    BNZ          ;7
      DW    BM1          ;8
      DW    BNM1         ;9
      DW    BK           ;A
      DW    RS           ;B
      DW    BS           ;C
      DW    NUL          ;D
      DW    NUL          ;E
      DW    NUL          ;F
      DW    NUL          ;0X  UNUSED
      DW    SET          ;1X
      DW    LD           ;2X
      DW    ST           ;3X
      DW    LDAT         ;4X
      DW    STAT         ;5X
      DW    LDDAT        ;6X
      DW    STDAT        ;7X
      DW    POP          ;8X
      DW    STPAT        ;9X
      DW    ADD          ;AX
      DW    SUB          ;BX
      DW    POPD         ;CX
      DW    CPR          ;DX
      DW    INR          ;EX
      DW    DCR          ;FX

LD      LDY  R0L,X
BK      EQU  *-1
        LDA  R0H,X          ;MOVE RX TO R0
STAY    STY  R0L
        STA  R0H
        RTS

ST      LDY  R0L
        LDA  R0H
        BCC  STZ

INR     STX  R14H
        BCC   INRZ

STAT    LDA  R0L
STAT2   STA  (R0L,X)        ;STORE BYTE INDIRECT
        JMP  INRZ  
         
LDAT    LDY  $0             ;ZERO HIGH ORDER R0 BYTE
LDATZ   JSR  LDAT2          ;LOAD INDIRECT (RX)
        JMP  STYA           ;TO R0

POP     LDY  $0             ;HIGH ORDER BYTE = 0
POP2    JSR  DCRZ           ;DECR RX  ;LOW ORDER BYTE
        LDA  (R0L,X) 
STYA    STA  R0L            ;TO R0
        STY  R0H
        RTS

POPD    JSR  DCRZ           ;DECR RX ;POP HIGH ORDER BYTE @RX
        LDA  (R0L,X) 
        TAY                 ;SAVE IN Y REG
        BCC  POP2

LDDAT   JSR  LDDAT2         ; WORD TO R0, INCR RX
        JMP  STAY

STDAT   JSR  STAT           ;STORE INDIRECT LOW ORDER
        LDA  R0H            ;BYTE AND INCR RX. THEN
        BCC  STAT2          ;STORE HIGH ORDER BYTE. ;INCR RX AND RETURN

STPAT   JSR  DCRZ           ;DECR RX
        LDA  R0L
        STA  (R0L,X)        ;STORE R0 LOW BYTE @RX
        RTS

DCR     STX  R14H
DCRZ    LDA  R0L,X
        BNE  DCR2           ;DECR RX  AND FETHCH BYTE @RX
        DEC  R0H,X
DCR2    DEC  R0L,X
        RTS

SUB     LDY  $0             ;RESULT TO R0
CPR     SEC                 ;NOTE Y REG = 13*2 FOR CPR
        LDA  R0L
        SBC  R0L,X
        STA  R0L,Y          ;R0-RX TO RY
        LDA  R0H
        SBC  R0H,X
SUB2    STA  R0H,Y
        TYA                 ;LAST RESULT REG*2
        ADC  $0             ;CARRY TO LSB
        STA  R14H
        RTS

ADD     LDA  R0L
        ADC  R0L,X
        STA  R0L            ;R0+RX TO R0
        LDA  R0H
        ADC  R0H,X
        LDY  $0             ;R0 FOR RESULT
        BEQ  SUB2           ;FINISH ADD

BS      JSR  DCRZ         ;PUSH LOW PC BYTE VIA R12
        LDA  R15H
        STA  (R0L,X)        ;STORE R0 LOW BYTE @RX
        JSR  DCRZ         ;PUSH LOW PC BYTE VIA R12
        LDA  R15L           ;NOTE X REG IS 12*2!
        STA  (R0L,X)        ;STORE R0 LOW BYTE @RX
BR      CLC
BNC     BCS  BNC2           ;NO CARRY TEST
BR1     JSR   GETBZ         ;DISPLACEMENT BYTE
        ORA   #0
        BPL  BR2
        DEY
BR2     ADC  R15L           ;ADD TO PC
        STA  R15L
        TYA
        ADC  R15H
        STA  R15H
BNC2    RTS

BC      BCS  BR
        RTS

BP      ASL                 ;DOUBLE RESULT-REG INDEX
        TAX                 ;TO X REG FOR INDEXING
        LDA  R0H,X          ;TEST FOR PLUS
        BPL  BR1            ;BRANCH IF SO
        RTS

BM      ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0H,X          ;TEST FOR MINUS
        BMI  BR1
        RTS

BZ      ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X          ;TEST FOR ZERO
        ORA  R0H,X          ;(BOTH BYTES)
        BEQ  BR1            ;BRANCH IF SO
        RTS

BNZ     ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X          ;TEST FOR NON-ZERO
        ORA  R0H,X          ;(BOTH BYTES)
        BNE  BR1            ;BRANCH IF SO
        RTS

BM1     ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X          ;CHECK BOTH BYTES
        AND  R0H,X          ;FOR $FF (MINUS 1)
        EOR  $FF
        BEQ  BR1            ;BRANCH IF SO
        RTS

BNM1    ASL                 ;DOUBLE RESULT-REG INDEX
        TAX
        LDA  R0L,X
        AND  R0H,X          ;CHECK BOTH BYTES FOR NO $FF
        EOR  $FF
        BNE  BR1            ;BRANCH IF NOT MINUS 1
NUL     RTS

RS      LDX  $18            ;12*2 FOR R12 AS STACK POINTER
        JSR  LDDAT2         ;DECR STACK POINTER
        STA  R15H
        STY  R15L
        RTS
